# 装依赖（Ubuntu 为例） sudo apt install -y build-essential cmake qt6-base-dev qt6-multimedia-dev \ libbullet-dev liblua5.4-dev libassimp-dev # 构建全 13 个程序 git clone https://github.com/Jdj497949/PhysicsEngine3D.git cd PhysicsEngine3D cmake -B build -DCMAKE_BUILD_TYPE=Release cmake --build build -j$(nproc) # 见证：15 关节人形布娃娃从 3 米跌落，15/15 关节完好、退出码 0 ./build/3DRagdollDemo 3
